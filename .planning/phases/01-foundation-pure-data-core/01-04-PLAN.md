    ```python
    # tests/conftest.py
    import sys


    def pytest_configure(config) -> None:
        """Fail-fast if anything importable from homelab_mcp leaks into the test process.

        Belt-and-suspenders for ruff TID251 (which does not recursively cover
        submodule imports — see RESEARCH Pitfall 4 + pyproject.toml comment).
        """
        leaked = [
            name for name in sys.modules
            if name == "homelab_mcp" or name.startswith("homelab_mcp.")
        ]
        if leaked:
            raise RuntimeError(
                f"Black-box rule violated: homelab_mcp modules in sys.modules at test start: {leaked}. "
                f"The framework must drive homelab-mcp via stdio_client only -- never import."
            )
    ```

    Use the Write tool. The file is plain Python — no async, no fixtures, just the hook function.

    Phase 4 will add session-scoped fixtures to this same conftest.py later. Plan 01's CONTEXT.md notes "tests/conftest.py owns the session-scoped fixtures from Phase 4." Phase 1 does not need any fixtures here yet — only the guard.
