```python
"""Slug + identifier helpers for the Phase 17 codegen.

Per Phase 17 CONTEXT.md decisions:
  - D-05: server_slug(name) normalizes serverInfo.name from the MCP initialize
    handshake. Rule: lowercase -> non-[a-z0-9] to underscore -> collapse runs
    of underscores -> strip leading/trailing underscores. Errors loud on empty
    or all-non-identifier input (callers in cli.py map this to an operator
    error per the FileNotFoundError pattern at cli.py:672-690).
  - Claude's Discretion (PascalCase helper): pascal_case + module_name handle
    tool-name -> class-name and tool-name -> file-name conversions with
    keyword guards (Pitfall 2 in 17-RESEARCH.md).

Pure stdlib (re + keyword). No external deps.
"""
from __future__ import annotations

import keyword
import re

_NON_IDENT = re.compile(r"[^a-zA-Z0-9_]+")


def server_slug(server_name: str) -> str:
    """Normalize an MCP server name into a directory slug.

    >>> server_slug("homelab-mcp")
    'homelab_mcp'
    >>> server_slug("My Server v2.0")
    'my_server_v2_0'
    >>> server_slug("___leading-trailing___")
    'leading_trailing'
    """
    if not isinstance(server_name, str):
        raise ValueError(f"server name must be str, got {type(server_name).__name__}")
    lowered = server_name.lower()
    underscored = _NON_IDENT.sub("_", lowered)
    collapsed = re.sub(r"_+", "_", underscored)
    stripped = collapsed.strip("_")
    if not stripped:
        raise ValueError(
            f"server name {server_name!r} normalizes to empty slug; cannot "
            "derive a generated/<slug>/ directory. The MCP server must "
            "report a non-empty serverInfo.name with at least one "
            "[a-z0-9] character after lowercasing."
        )
    return stripped


def pascal_case(tool_name: str) -> str:
    """Convert a tool name to a PascalCase class identifier.

    >>> pascal_case("create_vm")
    'CreateVm'
    >>> pascal_case("list-registered-servers")
    'ListRegisteredServers'
    >>> pascal_case("class")
    'Class_'
    >>> pascal_case("2nd_attempt")
    '_2ndAttempt'
    """
    parts = [p for p in _NON_IDENT.split(tool_name) if p]
    if not parts:
        raise ValueError(f"tool name {tool_name!r} has no identifier characters")
    pascal = "".join(p[:1].upper() + p[1:] for p in parts)
    if pascal[0].isdigit():
        pascal = "_" + pascal
    # keyword guard (rare for PascalCase but cheap insurance)
    if keyword.iskeyword(pascal) or keyword.iskeyword(pascal.lower()):
        pascal = pascal + "_"
    return pascal


def module_name(tool_name: str) -> str:
    """Convert a tool name to a snake_case module file name.

    >>> module_name("create_vm")
    'create_vm'
    >>> module_name("list-registered-servers")
    'list_registered_servers'
    >>> module_name("import")
    'import_'
    >>> module_name("2nd-attempt")
    '_2nd_attempt'
    """
    name = _NON_IDENT.sub("_", tool_name).strip("_").lower()
    if not name:
        raise ValueError(f"tool name {tool_name!r} has no identifier characters")
    if name[0].isdigit():
        name = "_" + name
    if keyword.iskeyword(name):
        name = name + "_"
    return name
```

```python
@dataclass(frozen=True)
class FieldSpec:
    """A single Pydantic-model field as the walker plans it.

    py_type:        the post-`from __future__ import annotations` annotation
                    string (e.g. "str", "list[int]", "typing.Any").

                    we don't degrade-then-nullable in v1.3 per D-02).
    degrade_reason: human-readable phrase if this field is a degradation,
                    else None. Emitted as `# codegen: degraded -- <reason>`
                    on the line above the field.
    """
    py_type: str
    default_expr: str
    degrade_reason: str | None


class SchemaValidityError(ValueError):
    """Raised when an MCP tool's inputSchema is not a valid JSON Schema document.

    The codegen refuses to translate an invalid schema (loud-fail per CONTEXT.md
    Reusable Assets); the CLI command (Plan 17-04) maps this to an operator
    error naming the offending tool.
    """


def _emit_field(name: str, schema: dict, *, required: bool) -> FieldSpec:
    """Translate one property's schema to a FieldSpec.

    Order of checks matters: degradation triggers (enum/oneOf/anyOf/$ref/
    multi-type) take precedence over `type:` dispatch so that
    `{"type":"string","enum":["a","b"]}` degrades to Any (not str) per D-02.
    """
    # Degradation triggers (D-02) -- check FIRST.
    if "enum" in schema:
        return FieldSpec("typing.Any", _default_expr(schema, required), "enum not in v1.3 coverage")
    for combinator in ("oneOf", "anyOf", "$ref"):
        if combinator in schema:
            return FieldSpec(
                "typing.Any", _default_expr(schema, required),
                f"{combinator} not in v1.3 coverage",
            )
    jtype = schema.get("type")
    if isinstance(jtype, list):
        return FieldSpec(
            "typing.Any", _default_expr(schema, required),
            "multi-type / nullable not in v1.3 coverage",
        )

    # Faithful translation paths (D-02).
    if isinstance(jtype, str) and jtype in _SCALAR_MAP:
        return FieldSpec(_SCALAR_MAP[jtype], _default_expr(schema, required), None)
    if jtype == "array":
        items = schema.get("items") or {}
        items_type = items.get("type")
        if isinstance(items_type, str) and items_type in _SCALAR_MAP:
            return FieldSpec(
                f"list[{_SCALAR_MAP[items_type]}]",
                _default_expr(schema, required),
                None,
            )
        return FieldSpec(
            "typing.Any", _default_expr(schema, required),
            "array with non-scalar items not in v1.3 coverage",
        )
    if jtype == "object":
        # Nested objects in v1.3: degrade to dict[str, Any]; full nested-model
        # emission is a v1.4 expansion. The current goal is the walker + the
        # SEAM, not deep type fidelity (D-02 narrow promise).
        return FieldSpec(
            "dict[str, typing.Any]", _default_expr(schema, required),
            None if "properties" not in schema else
            "nested-object recursion not in v1.3 coverage",
        )

    # Unknown / missing type.
    return FieldSpec(
        "typing.Any", _default_expr(schema, required),
        f"unknown type: {jtype!r}",
    )


def _default_expr(schema: dict, required: bool) -> str:
    """Build the `Field(...)` right-hand side for a single field.

    Required fields get `Field(...)` (Ellipsis). Optional fields with a
    declared default get `Field(default=<repr>)`. Optional fields without
    a default get `Field(default=None)` -- and the type gets a `| None`
    union applied at emit-site (handled in _format_field_line).

    description is lifted from schema["description"] if present; quoted via
    repr() for safe escaping.
    """
    desc = schema.get("description")
    desc_kwarg = f", description={desc!r}" if isinstance(desc, str) and desc else ""
    if required:
        return f"Field(...{desc_kwarg})"
    if "default" in schema:
        return f"Field(default={schema['default']!r}{desc_kwarg})"
    return f"Field(default=None{desc_kwarg})"


def _format_field_line(name: str, spec: FieldSpec, *, required: bool) -> str:
    """Render a FieldSpec to source lines (degrade comment + field decl)."""
    py_type = spec.py_type
    # Optional fields without an explicit default get `| None`. We add this
    # only for faithful scalar paths -- degradation already types to Any.
    if not required and "default=None" in spec.default_expr and spec.degrade_reason is None and py_type != "typing.Any":
        py_type = f"{py_type} | None"
    lines: list[str] = []
    if spec.degrade_reason is not None:
        lines.append(f"    # codegen: degraded -- {spec.degrade_reason}")
    lines.append(f"    {name}: {py_type} = {spec.default_expr}")
    return "\n".join(lines)


def _emit_params_body(schema: dict | None) -> tuple[str, int]:
    """Emit the body of a <Tool>Params class. Returns (body_text, degraded_count).

    Empty / non-object inputSchema -> `pass` body. degraded_count is 0 in that case.
    """
    if not isinstance(schema, dict) or schema.get("type") != "object":
        return "    pass\n", 0
    props = schema.get("properties") or {}
    required = set(schema.get("required") or [])
    if not props:
        return "    pass\n", 0
    degraded = 0
    lines: list[str] = []
    # Required first (alphabetical), then optional (alphabetical) -- matches
    # `_format_param_signature` at cli.py:932-950.
    req_names = sorted(n for n in props if n in required)
    opt_names = sorted(n for n in props if n not in required)
    for name in req_names + opt_names:
        prop_schema = props[name] if isinstance(props[name], dict) else {}
        spec = _emit_field(name, prop_schema, required=(name in required))
        if spec.degrade_reason is not None:
            degraded += 1
        lines.append(_format_field_line(name, spec, required=(name in required)))
    return "\n".join(lines) + "\n", degraded


def _emit_response_body(schema: dict | None) -> tuple[str, int]:
    """Emit the body of a <Tool>Response class. D-06 stub when schema is None."""
    if schema is None:
        return "    pass\n", 0
    # When outputSchema IS declared, treat it the same as Params (object walk).
    return _emit_params_body(schema)


def translate_tool(
    tool: Tool,
    *,
    slug: str,
    server_name: str,
    server_version: str,
    timestamp: str,
) -> tuple[str, int]:
    """Translate one MCP Tool to a per-tool source file (str). Pure-data.

    Returns (source_text, degraded_field_count).

    Refuses to translate if `tool.inputSchema` is not a valid JSON Schema
    document (raises SchemaValidityError per "Don't Hand-Roll" row 1).
    """
    input_schema = tool.inputSchema if isinstance(tool.inputSchema, dict) else {}
    if input_schema:
        try:
            validator_cls = validator_for(input_schema, default=Draft202012Validator)
            validator_cls.check_schema(input_schema)
        except Exception as exc:  # jsonschema.exceptions.SchemaError
            raise SchemaValidityError(
                f"tool {tool.name!r}: inputSchema is not a valid JSON Schema "
                f"document ({exc.__class__.__name__}: {exc}). Refusing to "
                f"translate."
            ) from exc

    cls_base = pascal_case(tool.name)
    header = HEADER_TEMPLATE.format(
        slug=slug, server_name=server_name,
        server_version=server_version, timestamp=timestamp,
    )

    params_body, params_degraded = _emit_params_body(input_schema)
    output_schema = tool.outputSchema if isinstance(tool.outputSchema, dict) else None
    response_body, response_degraded = _emit_response_body(output_schema)

    response_class = (
        f"class {cls_base}Response(ToolResponse):\n"
        f'    """Typed response for the `{tool.name}` tool.\n\n'
        f"    Inherits .raw / .data / .text / .is_error from ToolResponse (CODEGEN-04).\n"
    )
    if output_schema is None:
        # D-06: stub class -- pass body, document the lack of outputSchema.
        response_class += (
            f"    outputSchema was undeclared at codegen time; .data falls\n"
            f"    back to JSON-parse / text-dict per CODEGEN-04 chain.\n"
            f'    """\n'
            f"{response_body}"
        )
    else:
        response_class += f'    """\n{response_body}'

    source = (
        f"{header}"
        f"from __future__ import annotations\n\n"
        f"import typing\n\n"
        f"from pydantic import BaseModel, ConfigDict, Field\n\n"
        f"from mcp_test_framework.sdet.response import ToolResponse\n\n\n"
        f"class {cls_base}Params(BaseModel):\n"
        f'    """Params for the `{tool.name}` tool.\n\n'
        f"    Generated from inputSchema. Pydantic validates required / type /\n"
        f"    default rules at construction time before the wire call.\n"
        f'    """\n'
        f"    model_config = ConfigDict(extra=\"forbid\")\n\n"
        f"{params_body}\n\n"
        f"{response_class}"
    )
    return source, params_degraded + response_degraded
```

```python
def _render_init(
    *,
    slug: str,
    server_name: str,
    server_version: str,
    timestamp: str,
    registry_entries: list[tuple[str, str]],  # [(tool_name, pascal_base), ...]

    """Render generated/<slug>/__init__.py: header + per-tool imports + _REGISTRY.

    Per RESEARCH Pattern 4 + CONTEXT.md "What gen-sdet-classes prints" (sorted).
    """
    header = HEADER_TEMPLATE.format(
        slug=slug, server_name=server_name,
        server_version=server_version, timestamp=timestamp,
    )
    # Imports, sorted alphabetically by tool_name (diff-stable).
    import_lines: list[str] = []
    all_names: list[str] = []
    registry_lines: list[str] = []

    `timestamp=None` uses datetime.now(UTC) at sub-second precision; tests
    pass a fixed value for byte-identical idempotency.

    Returns counts: {"tools": <N>, "degraded_fields": <M>}.
    """
    if timestamp is None:
        timestamp = _dt.datetime.now(_dt.UTC).isoformat(timespec="seconds")

    slug = server_slug(server_name)  # raises ValueError on empty/all-non-ident
    target = out_root / slug

    # D-03 wipe.
    shutil.rmtree(target, ignore_errors=True)
    target.mkdir(parents=True, exist_ok=True)

    counts = {"tools": 0, "degraded_fields": 0}
    registry_entries: list[tuple[str, str]] = []

    # Sorted emission for diff-stability.
    for tool in sorted(tools, key=lambda t: t.name):
        rendered, degraded_n = translate_tool(
            tool, slug=slug, server_name=server_name,
            server_version=server_version, timestamp=timestamp,
        )
        per_tool_path = target / f"{module_name(tool.name)}.py"
        per_tool_path.write_text(rendered, encoding="utf-8")
        counts["tools"] += 1
        counts["degraded_fields"] += degraded_n
        registry_entries.append((tool.name, pascal_case(tool.name)))

    # Write __init__.py LAST (Pitfall 5).
    init_text = _render_init(
        slug=slug, server_name=server_name,
        server_version=server_version, timestamp=timestamp,
        registry_entries=registry_entries,
    )
    (target / "__init__.py").write_text(init_text, encoding="utf-8")

    return counts
```
