def _run_pytest_with_tools_yaml(
    tmp_path: Path,
    tools_block: dict,
    extra_args: list[str] | None = None,
) -> subprocess.CompletedProcess[str]:
    """Helper: drive `pytest tests/test_mcp_tool_contract.py` under a tempfile
    YAML overlay built by DEEP-MERGING `tools_block` into config.example.yaml.

    Why deep-merge instead of string concatenation: after Plan 04 Task 3 lands,
    config.example.yaml itself contains a top-level `tools:` block. Naive
    concatenation (`base_text + "\\n" + tools_yaml + "\\n"`) produces duplicate
    top-level `tools:` keys -- PyYAML's safe_load silently keeps the last,
    which means tests pass today but break invisibly the moment loader behavior
    changes or a third-party tool reads the YAML strictly. Deep-merge via
    yaml.safe_load + dict.update keeps the file syntactically valid AND
    preserves the example file's other settings (mcp_server, ollama, target,
    judge_timeout_seconds), which is what we want -- the test inherits the
    operator's mcp_server config from the canonical example.

    Args:
        tools_block: a dict shaped like {"<tool_name>": {<ToolConfig fields>}}
                     to merge under the YAML's top-level `tools:` key.
    """
    # Parse the canonical example as the base (it now contains version: 1 +
    # a worked tools: block per Plan 04 Task 3).
    base_doc = yaml.safe_load(
        Path("config.example.yaml").read_text(encoding="utf-8")
    ) or {}

    # Deep-merge: ensure base_doc has a tools: dict, then update it with the
    # test's entries (test entries OVERRIDE example entries when names collide,
    # which is the desired semantics for runtime tests targeting a specific tool).
    merged_tools = dict(base_doc.get("tools") or {})
    merged_tools.update(tools_block)
    base_doc["tools"] = merged_tools

    yaml_path = tmp_path / "tools_overlay.yaml"
    yaml_path.write_text(
        yaml.safe_dump(base_doc, sort_keys=False, default_flow_style=False),
        encoding="utf-8",
    )

    env = {**os.environ, "MCPTF_CONFIG_FILE": str(yaml_path)}
    cmd = [
        sys.executable, "-m", "pytest",
        "tests/test_mcp_tool_contract.py",
        "-v",
        "-W", "default::UserWarning",
        # Don't filter out live_* markers -- this subprocess invocation is
        # itself the live check, gated by the outer test's @live_* markers.
        "-m", "",
    ]
    if extra_args:
        cmd.extend(extra_args)
    return subprocess.run(cmd, env=env, capture_output=True, text=True, timeout=300)
