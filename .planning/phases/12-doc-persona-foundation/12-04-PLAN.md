```python
def _emit_operator_error(
    summary: str,
    detail: list[str],
    next_step: str,
    *,
    exit_code: int = 2,
) -> typer.Exit:
    """Render an operator-grade error and exit (citation: docs/ERROR-STYLE.md).

    Format (per docs/ERROR-STYLE.md):
        <one-line summary>
        <blank>
        <detail line 1>
        <detail line 2>
        ...
        <blank>
        next: <action verb> <command-or-instruction>

    Operator terms only -- no spec IDs, no file:line refs, no internal jargon.
    Caller is expected to `raise` the return value; the function itself
    raises typer.Exit so the type-checker sees a NoReturn-style flow.
    """
    parts: list[str] = [summary, ""]
    parts.extend(detail)
    parts.extend(["", f"next: {next_step}"])
    typer.echo("\n".join(parts), err=True)
    raise typer.Exit(code=exit_code)
```

def _emit_operator_error_for_validation(exc: ValidationError, *, source: str) -> None:
    """Map pydantic.ValidationError -> operator-tone error per docs/ERROR-STYLE.md.

    Mapping rules:
    - version mismatch (config version N not supported by this build, expected 1)
        -> "config file uses an older format" framing (forward-compat with SAFE-06
           reference message in docs/ERROR-STYLE.md, but Phase 12 still accepts v1
           and rejects v2+; the message names the actual mismatch).
    - extra_forbidden on `target.tool_name` -> v1.2 deprecation hint
    - missing required field -> point at config.example.yaml
    - other validation errors -> generic detail block with the field path
    """
    errors = exc.errors()
    # Detect the most useful single error and write the message for it; the
    # remaining errors are summarised in the detail block.
    primary = errors[0] if errors else {}
    loc = ".".join(str(p) for p in primary.get("loc", ()))
    err_type = primary.get("type", "")
    msg = primary.get("msg", "")

    if loc == "version" and "not supported by this build" in msg:
        # Reuse the SAFE-06-shape framing (operator-tone), but Phase 12 only
        # accepts v1, not v2 -- the message matches the actual contract.
        _emit_operator_error(
            summary=f"config file uses an unsupported schema version: {source}",
            detail=[
                "this release of mcp-test-framework accepts schema version 1.",
                f"the file declares: {msg}.",
                "",
                "regenerate a starter file and port your tool entries across.",
            ],
            next_step=(
                "run `mcp-test-framework config-init -o config.yaml.new` to see "
                "the expected layout, then merge your tool entries into it"
            ),
        )
    if err_type == "extra_forbidden" and "tool_name" in loc:
        _emit_operator_error(
            summary=f"config file uses a removed field: {loc}",
            detail=[
                "the `target.tool_name` field was removed in v1.2.",
                "the framework now uses the `tools:` block to decide which tools run.",
                "",
                "remove the `target.tool_name` line (and the `target:` block if "
                "it is now empty) from your config file.",
            ],
            next_step=(
                "edit your config file or run "
                "`mcp-test-framework config-init -o config.yaml` to regenerate"
            ),
        )
    if err_type in ("missing", "value_error.missing"):
        _emit_operator_error(
            summary=f"config file is missing a required field: {loc}",
            detail=[
                f"the field `{loc}` is required but was not found in {source}.",
                "",
                "see config.example.yaml for the expected shape, or regenerate "
                "a starter file with config-init.",
            ],
            next_step=(
                "copy the relevant block from config.example.yaml or run "
                "`mcp-test-framework config-init -o config.yaml`"
            ),
        )
    # Generic fallback -- still operator-tone, no pydantic-internal terms.
    field_summary = ", ".join(
        ".".join(str(p) for p in e.get("loc", ())) for e in errors
    ) or "(unknown field)"
    detail_lines = [f"the following config field(s) failed validation: {field_summary}."]
    for e in errors[:3]:
        detail_lines.append(f"  - {'.'.join(str(p) for p in e.get('loc', ()))}: {e.get('msg', '')}")
    _emit_operator_error(
        summary=f"config file has invalid values: {source}",
        detail=detail_lines,
        next_step=(
            "fix the field(s) above, or run "
            "`mcp-test-framework config-init -o config.yaml` to regenerate"
        ),
    )
```

def _format_tools_yaml_scaffold(tools: list[Tool]) -> str:
    """Hand-format a complete self-contained YAML scaffold (CLEAN-05).

    Output shape (locked in docs/ERROR-STYLE.md / CLEAN-05 acceptance):
    - Top-level `ollama:`, `mcp_server:`, `judge_timeout_seconds:`, `version:`,
      `tools:` blocks all populated.
    - `version: 1` literal (this release's accepted version).
    - Every discovered tool emitted as `<name>: { skip: true, skip_reason: ... }`.
    - No `target:` block (the field is leaving in a future schema bump).
    - All string scalars JSON-quoted via `json.dumps` (avoids YAML scalar
      ambiguity for values containing colons, hashes, or quotes).

    Tools sorted alphabetically (preserves the v1.1 D-list-4 contract).
    """
    sorted_tools = sorted(tools, key=lambda t: t.name)
    skip_reason = "review and remove skip to enable"

    header = (
        "# mcp-test-framework starter config -- generated by `config-init`.\n"
        "# Edit this file in place, or copy to a project-local path and pass --config PATH.\n"
        "\n"
        "# Ollama judge backend.\n"
        "ollama:\n"
        f"  base_url: {json.dumps('http://127.0.0.1:11434')}\n"
        f"  model: {json.dumps('qwen3.6:latest')}\n"
        "  timeout_seconds: 120\n"
        "\n"
        "# MCP server under test.\n"
        "# The launch command MUST be runnable from your shell.\n"
        "mcp_server:\n"
        f"  command: {json.dumps('uvx')}\n"
        f"  args: [{json.dumps('your-mcp-server-package')}]\n"
        "  timeout_seconds: 30\n"
        "\n"
        "# Outer budget cap on each judge HTTP call.\n"
        "judge_timeout_seconds: 120\n"
        "\n"
        "# Schema version. This release accepts version 1.\n"
        "version: 1\n"
        "\n"
        "# Per-tool registry. Every tool the connected server advertises is\n"
        "# listed below as `skip: true` -- the framework will not call any\n"
        "# tool until you review and remove `skip` from the ones you want\n"
        "# to test.\n"
        "#\n"
        "# Read each entry below, decide whether it is safe to call against\n"
        "# your environment, then either:\n"
        "#   - delete the `skip` and `skip_reason` lines to enable the tool, OR\n"
        "#   - leave the entry as-is to keep the tool out of the test surface.\n"
        "tools:\n"
    )

    if not sorted_tools:
        return header + "  {}\n"

    blocks: list[str] = []
    for tool in sorted_tools:
        block = (
            f"  {tool.name}:\n"
            f"    skip: true\n"
            f"    skip_reason: {json.dumps(skip_reason)}\n"
        )
        blocks.append(block)

    return header + "\n".join(blocks) + "\n"
```

Keep `f"  {tool.name}:"` (un-quoted) since MCP tool names per the spec match `^[a-zA-Z][a-zA-Z0-9_]*$`. If a name violates that, json.dumps it. Inline the rule:

```python
def _yaml_key(name: str) -> str:
    """Return name as a YAML mapping key, JSON-quoted only if necessary."""
    import re
    if re.fullmatch(r"[A-Za-z_][A-Za-z0-9_]*", name):
        return name
    return json.dumps(name)
```

import os
from pathlib import Path
from typing import Any

import pytest
import yaml

from mcp_test_framework.cli import _format_tools_yaml_scaffold

class _StubTool:

def _scaffold(tool_names: list[str]) -> str:
    return _format_tools_yaml_scaffold([_StubTool(n) for n in tool_names])


def _parse(yaml_text: str) -> dict[str, Any]:
    data = yaml.safe_load(yaml_text)
    assert isinstance(data, dict)
    return data


def test_scaffold_has_top_level_ollama() -> None:
    data = _parse(_scaffold(["a"]))
    assert "ollama" in data
    o = data["ollama"]
    assert "base_url" in o and "model" in o and "timeout_seconds" in o

    data = _parse(_scaffold(["a"]))
    assert "mcp_server" in data
    s = data["mcp_server"]
    assert "command" in s and "args" in s and "timeout_seconds" in s

    assert isinstance(data["judge_timeout_seconds"], int)

    assert "target" not in data


def test_scaffold_tools_all_skip_true() -> None:

    data = _parse(_scaffold(["alpha", "beta", "gamma"]))
    tools = data["tools"]
    assert set(tools) == {"alpha", "beta", "gamma"}
    for name, entry in tools.items():
        assert entry["skip"] is True, name
        assert entry["skip_reason"] == "review and remove skip to enable", name

    assert "tools:\n  {}\n" in text or "tools: {}\n" in text
    data = yaml.safe_load(text)
    assert data["tools"] in (None, {})


def test_scaffold_loadable_via_config(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> None:
    """CLEAN-05 acceptance: the scaffold loads via Config() with no .env present."""
    # Aggressively clear any env-overlay vars that could backfill missing fields
    for var in (
        "OLLAMA_BASE_URL", "OLLAMA_MODEL", "OLLAMA_TIMEOUT_SECONDS",
        "MCP_SERVER_COMMAND", "MCP_SERVER_ARGS", "MCP_SERVER_TIMEOUT_SECONDS",
        "JUDGE_TIMEOUT_SECONDS", "TARGET_TOOL_NAME", "MCPTF_CONFIG_FILE",
    ):
        monkeypatch.delenv(var, raising=False)
    cfg_path = tmp_path / "config.yaml"
    cfg_path.write_text(_scaffold(["alpha", "beta"]), encoding="utf-8")
    monkeypatch.setenv("MCPTF_CONFIG_FILE", str(cfg_path))

    cfg = Config()
    assert cfg.version == 1
    assert cfg.mcp_server.command == "uvx"
    assert cfg.ollama.base_url.endswith("11434") or "11434" in str(cfg.ollama.base_url)


def test_scaffold_no_banned_tokens() -> None:
    import re
    text = _scaffold(["alpha", "beta"])
    banned = [
        r"\bPhase \d", r"\bPlan \d-\d", r"\bTOOLCFG-\d", r"\bISOL-\d",
        r"\bCD-\d", r"\bD-\d{2}", r"\b\d{6}-[a-z0-9]{3}",
        r"src/[\w/.]+\.py:\d+",
    ]
    for p in banned:
        assert not re.search(p, text), f"banned pattern {p!r} in scaffold output"


def test_scaffold_sorted_alphabetically() -> None:
    """Preserve the v1.1 D-list-4 sort contract."""
    text = _scaffold(["zeta", "alpha", "mu"])
    lines = text.splitlines()
    tool_lines = [l for l in lines if l.startswith("  ") and l.endswith(":") and not l.startswith("  #")]
    # Expect alpha, mu, zeta order
    names = [l.strip().rstrip(":") for l in tool_lines if l.strip().rstrip(":") in {"alpha", "mu", "zeta"}]
    assert names == ["alpha", "mu", "zeta"]

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| operator → cli.py error path | Trusted input (operator's own --config path); errors do not echo back arbitrary user-controlled content beyond the path string |
| MCP server → config-init scaffold | Untrusted: tool names come from the live MCP server. A malicious server could declare a tool name containing YAML control characters trying to break the emitted scaffold |
| operator → config-init output file | Operator opens the generated YAML; any injected control characters could mislead the operator about what will be skipped |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-12-09 | Tampering | _format_tools_yaml_scaffold | mitigate | A malicious MCP server emitting `tool.name = "alpha:\n  skip: false\n# beta"` could try to inject opt-in for a separate tool. Mitigation: `_yaml_key(name)` JSON-quotes any name not matching `^[A-Za-z_][A-Za-z0-9_]*$`; all string scalars (skip_reason, base_url, model, command, args items) JSON-quoted via `json.dumps`. JSON-quoted strings are always valid YAML 1.2 scalars and cannot break out of the value position. |
| T-12-10 | Information Disclosure | _emit_operator_error | mitigate | Error messages echo the operator's `--config` path verbatim. The path may be an absolute filesystem path. The banned-token grep allows `src/.*\.py:\d+` to match in tests but the actual error helper does not emit file:line refs — only operator-supplied paths. Acceptable: the operator already knows their own filesystem layout. The error does NOT echo env-var values, file contents, or pydantic schema internals (those are mapped to operator-tone wording). |
| T-12-11 | Denial of Service | _emit_operator_error_for_validation | accept | A pathological config with thousands of validation errors could produce a long error message. Mitigated by `errors[:3]` cap in the generic-fallback branch. |
| T-12-12 | Information Disclosure | _emit_operator_error_for_validation | mitigate | Pydantic ValidationError text can include field values. The mapping rules use `loc` (field path) and `msg` (validator message) — never `input` or `ctx` which can echo the bad value. The "removed field" branch (`extra_forbidden` on `target.tool_name`) intentionally tells the operator the field name to delete; not a sensitive disclosure. |
